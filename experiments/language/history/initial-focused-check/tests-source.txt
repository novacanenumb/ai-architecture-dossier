import test from 'node:test';
import assert from 'node:assert/strict';
import {analyseLanguageWindow as analyse,compareLanguageWindows as compare} from '../packages/lab/language-analysis.mjs';
const input=(text='a b a c',extra={})=>({id:'a',arm:'baseline',topicId:'topic',taskId:'task',participant:{id:'synthetic:one',kind:'synthetic'},segments:[{kind:'authored',text}],...extra});
const options={sampleTokens:4,mattrWindow:3};
const pair=(left='red red red red',right=left,extra={})=>[analyse(input(left),options),analyse(input(right,{id:'b',arm:'candidate',...extra}),options)];
test('C14-T01 fixed authored sample matches known MATTR and retains configuration',()=>{const r=analyse(input(),options);assert.equal(r.status,'READY');assert.equal(r.authored.mattr.value,5/6);assert.deepEqual(r.authored.tokensSample,['a','b','a','c']);assert.deepEqual(r.measurementConfig,options);assert.match(r.authored.mattr.convention,/pooled/);});
test('C14-T01 short unique text remains insufficient with null metrics',()=>{const r=analyse(input('alpha beta gamma'),options);assert.equal(r.eligible,false);assert.equal(r.counts.authored,3);for(const key of ['mattr','repeatedTrigramRate','sentenceLengthCV'])assert.equal(r.authored[key].value,null);});
test('C14-T01 fixed sample keeps first words and reports truncation',()=>{const r=analyse(input('one two three four five six'),options);assert.equal(r.counts.authored,6);assert.deepEqual(r.authored.tokensSample,['one','two','three','four']);assert.equal(r.sampling.truncated,true);});
test('C14-T02 quotations and required terms remain separate from authored metrics',()=>{const base=input();base.segments.push({kind:'quotation',text:'quoted words',sourceId:'synthetic:quote'},{kind:'required_term',text:'sealed boundary'});const longer=structuredClone(base);longer.segments[1].text='quoted words with many many more quoted words';const a=analyse(base,options),b=analyse(longer,options);assert.deepEqual(a.authored,b.authored);assert.equal(a.counts.required_term,2);assert.ok(b.counts.quotation>a.counts.quotation);assert.equal(a.annotations.quotationSourceIdsIndependentlyVerified,false);});
test('C14-T02 short authored segments manufacture no cross-boundary trigrams',()=>{const r=analyse(input('',{segments:[{kind:'authored',text:'alpha beta'},{kind:'quotation',text:'excluded quote',sourceId:'synthetic:quote'},{kind:'authored',text:'gamma delta'}]}),options);assert.equal(r.authored.repeatedTrigramRate.value,null);assert.equal(r.authored.repeatedTrigramRate.denominator,0);});
test('C14 known identical and disjoint lexical supports produce JSD zero and one',()=>{const [a,b]=pair();assert.equal(compare(a,b).lexicalJSD.value,0);const [c,d]=pair('red red red red','blue blue blue blue');const r=compare(c,d);assert.equal(r.lexicalJSD.value,1);assert.deepEqual(r.lexicalJSD.support,['blue','red']);assert.equal(r.lexicalJSD.unit,'bits');});
test('C14-T03 topic and task mismatches refuse derived differences',()=>{for(const extra of [{topicId:'other'},{taskId:'other'}]){const r=compare(...pair(undefined,undefined,extra));assert.equal(r.eligible,false);assert.ok(r.reasons.length);assert.equal(r.lexicalJSD.value,null);assert.equal(r.authoredMATTRDifference.value,null);}});
test('C14-T01 metric configuration and record identity mismatch are ineligible',()=>{const a=analyse(input(),options);for(const b of [analyse(input(undefined,{id:'b'}),{sampleTokens:4,mattrWindow:2}),analyse(input(),options)])assert.equal(compare(a,b).eligible,false);});
test('C14-T04 synthetic model and human descriptors cannot be pooled',()=>{for(const kind of ['model','human'])assert.equal(compare(...pair(undefined,undefined,{participant:{id:'other',kind}})).eligible,false);const a=analyse(input(),options);assert.equal(a.synthetic,true);assert.equal(a.participant.kind,'synthetic');});
test('C14-T04 human descriptors never manufacture human or semantic outcomes',()=>{const [a,b]=pair();a.participant.kind=b.participant.kind='human';a.synthetic=b.synthetic=false;const r=compare(a,b);assert.equal(r.eligible,true);for(const key of ['humanOutcome','semanticAlignment','transferTaskScore'])assert.equal(r[key],null);});
test('C14 Unicode words and internal apostrophes survive fixed tokenization',()=>{const r=analyse(input("CAFÉ Don't naïve ÉLAN"),options);assert.deepEqual(r.authored.tokensSample,['café',"don't",'naïve','élan']);});
test('C14 invalid kinds quotation annotation and bounds fail explicitly',()=>{for(const segments of [[{kind:'unsupported',text:'a b c d'}],[{kind:'quotation',text:'a b c d'}]])assert.throws(()=>analyse(input('',{segments}),options));for(const invalid of [{sampleTokens:0,mattrWindow:3},{sampleTokens:4,mattrWindow:0},{sampleTokens:4,mattrWindow:5},{sampleTokens:4097,mattrWindow:3}])assert.throws(()=>analyse(input(),invalid));});
test('C14 analysis and comparison preserve caller inputs and options',()=>{const a=input(),b=input(undefined,{id:'b',arm:'candidate'}),snapshot=structuredClone([a,b,options]);compare(analyse(a,options),analyse(b,options));assert.deepEqual([a,b,options],snapshot);});
test('C14 forged MATTR invalid counts and malformed tokens cannot yield comparison',()=>{const [a,b]=pair();for(const change of [r=>{r.authored.mattr.value=NaN;},r=>{r.authored.mattr.value=.99;},r=>{r.counts.whole++;},r=>{r.authored.tokensSample[0]='two words';}]){const altered=structuredClone(b);change(altered);const r=compare(a,altered);assert.equal(r.eligible,false);assert.equal(r.lexicalJSD.value,null);}});
test('C14 truncated sentence does not become a complete rhythm observation',()=>{const r=analyse(input('one two. three four five six.'),options);assert.equal(r.authored.sentenceLengthCV.value,null);assert.equal(r.authored.sentenceLengthCV.observations,1);assert.deepEqual(r.authored.sentenceLengthCV.sentenceTokenLengths,[2]);});
test('C14 repeated trigram count retains real repetition across distinct segments',()=>{const r=analyse(input('',{segments:[{kind:'authored',text:'a b c'},{kind:'authored',text:'a b c'}]}),{sampleTokens:6,mattrWindow:3});assert.equal(r.authored.repeatedTrigramRate.numerator,1);assert.equal(r.authored.repeatedTrigramRate.denominator,2);assert.equal(r.authored.repeatedTrigramRate.value,.5);});
