function nativeArchivePanel(componentId) {
  if(componentId&&componentId!=='C25')return '';
  if(!nativeArchiveEvidence||!nativeArchiveResults)return '<p>Recorded native archive evidence unavailable.</p>';
  const r=nativeArchiveResults,t=r.localInferenceTime,j=r.returnedJson;
  return `<h2>Native archive representation and rejection evidence</h2><p>30 cases passed: 26 unchanged original cases and four new probes. Sixteen source inputs were pinned and rechecked unchanged. All eight paired cases preserve shared distribution, CRSC record, collapse, frame and event hashes, with read-only reopen parity.</p><div class="scroll"><table><thead><tr><th>Local fixture measure</th><th>Full capture</th><th>Sealed archive</th></tr></thead><tbody><tr><td>Mean infer_token time</td><td>${t.baselineMean.toFixed(4)} ms</td><td>${t.candidateMean.toFixed(4)} ms</td></tr><tr><td>Returned JSON</td><td>${j.baselineMean} bytes</td><td>${j.candidateMean} bytes</td></tr><tr><td>Stored objects, metadata and timing sidecar</td><td>Unavailable: no disk artifact</td><td>${r.storedArchive.candidateMeanBytes} bytes</td></tr></tbody></table></div><p>Returned JSON is ${(100*j.reduction.value).toFixed(2)}% smaller, while timed local inference is ${t.candidateToBaselineRatio.value.toFixed(2)} times longer. The candidate adds disk writes and sealing. The paired timing difference is ${t.bootstrap.meanDifference.toFixed(4)} ms; 95% seeded bootstrap interval [${t.bootstrap.interval.map(v=>v.toFixed(4)).join(', ')}] ms. All 8 pairs remain in the denominator; 0 excluded. Two warmup arms are recorded separately.</p><p>Timing covers inference and its in-call capture/hashing/seal, excluding allocation, directory preparation, expansion, comparison hashes, reopen and byte enumeration. Both arms emit telemetry. JSON transport and added disk storage have separate accounting; no disk compression ratio is claimed.</p><p>C25-T04 has partial missing-generated-object rejection evidence. General CRSC recovery, C25-T03 eviction and C26-T05 telemetry-off overhead remain unverified. These synthetic fixtures do not measure trained model quality, provider latency or full Hypervisor improvement.</p><div class="actions"><a class="button" href="./data/native-archive-evidence.json" download>Download original cases and sanitized observations</a><a class="button" href="./data/native-archive-results.json" download>Download paired statistics</a></div><p>Reproduce: <code>npm run test:native-archive</code>. Statistics only: <code>npm run derive:native-archive</code>. All hosted endpoints remain null.</p>`;
}
function renderArchiveCase(pairId,family) {
  if(!['all','original','probe'].includes(family))return '<p>Recorded archive test family unavailable.</p>';
  const pair=nativeArchiveResults.pairs.find(p=>String(p.pair)===String(pairId));
  if(!pair)return '<p>Recorded archive pair unavailable.</p>';
  const arms=nativeArchiveEvidence.summary.arms.filter(a=>a.pair===pair.pair);
  const cases=nativeArchiveEvidence.tests.results.filter(t=>family==='all'||(t.id.startsWith('probe.py::')?(family==='probe'):(family==='original')));
  return `<h3>Recorded archive pair ${pair.pair}</h3><p>Shared hashes and reopened parity: ${pair.eligible?'MATCH':'UNAVAILABLE / FAILED'}. ${cases.length} recorded cases; ${cases.filter(t=>t.status==='PASSED').length} passed.</p><div class="scroll"><table><thead><tr><th>Arm in recorded order</th><th>Infer time</th><th>JSON bytes</th><th>Disk bytes</th><th>Reopen hash gate</th></tr></thead><tbody>${arms.map(a=>`<tr><td>${escape(a.mode)}</td><td>${a.durationMs} ms</td><td>${a.returnedJsonBytes}</td><td>${a.storedArchiveAndTimingBytes??'Unavailable'}</td><td>${a.reopenedSharedHashesEqual===null?'Baseline full capture':a.reopenedSharedHashesEqual?'MATCH':'FAILED'}</td></tr>`).join('')}</tbody></table></div><details><summary>Inspect shared field hashes</summary><pre>${escape(JSON.stringify(arms.map(a=>({mode:a.mode,sharedHashes:a.sharedHashes})),null,2))}</pre></details><ul>${cases.map(t=>`<li><code>${escape(t.id)}</code> — ${escape(t.status)}</li>`).join('')}</ul>`;
}
function nativeArchiveControls() {
  return `<span class="badge">FROZEN NATIVE ARCHIVE SOURCE EXECUTION</span><label>Recorded pair<select id="archive-pair">${Array.from({length:8},(_,i)=>`<option value="${i}">Pair ${i} / ${i%2?'archive first':'full first'}</option>`).join('')}</select></label><label>Recorded test family<select id="archive-case-family"><option value="all">All 30 cases</option><option value="original">Original 26 cases</option><option value="probe">New probes 4 cases</option></select></label><button id="archive-inspect">Inspect recorded pair and cases</button><div id="archive-result" aria-live="polite">${renderArchiveCase(0,'all')}</div>${nativeArchivePanel('C25')}`;
}
function mountNativeArchiveLab() {
  app.querySelector('#archive-inspect').onclick=()=>{app.querySelector('#archive-result').innerHTML=renderArchiveCase(app.querySelector('#archive-pair').value,app.querySelector('#archive-case-family').value);};
}
